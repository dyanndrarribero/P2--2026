
#  COMANDOS PARA COMPILAR E EXECUTAR O SISTEMA DE STREAMING





# ── OPÇÃO 1: usando Makefile (recomendado) 

# Compilar
make

# Compilar e executar em seguida
make run

# Remover executável e arquivos .bin gerados
make clean


# ── OPÇÃO 2: compilação manual com gcc

# Compilar (Linux / macOS)
gcc -Wall -Wextra -std=c11 -o streaming main.c usuarios.c videos.c

# Executar
./streaming

# Compilar (Windows com MinGW)
gcc -Wall -Wextra -std=c11 -o streaming.exe main.c usuarios.c videos.c

# Executar (Windows)
streaming.exe


# ARQUIVOS GERADOS DURANTE A EXECUÇÃO 
#   usuarios.bin       → dados dos usuários
#   videos.bin         → dados dos vídeos
#   (arquivos temporários são criados e removidos automaticamente
#    durante operações de exclusão)
